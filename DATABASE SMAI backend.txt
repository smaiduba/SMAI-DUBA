/**
 * Konversi huruf kolom ke index (0-based)
 * Contoh: 'A' -> 0, 'Z' -> 25, 'AA' -> 26
 */
function getColumnIndex(columnLetter) {
  var letter = String(columnLetter).toUpperCase();
  var index = 0;
  for (var i = 0; i < letter.length; i++) {
    index = index * 26 + (letter.charCodeAt(i) - 64);
  }
  return index - 1;
}

/**
 * Konversi angka ke huruf terbilang (Bahasa Indonesia)
 */
function angkaTerbilang(angka) {
  if (!angka && angka !== 0) return '';
  var num = Math.round(angka);
  if (num === 0) return 'Nol';

  var satuan = ['', 'Satu', 'Dua', 'Tiga', 'Empat', 'Lima', 'Enam', 'Tujuh', 'Delapan', 'Sembilan'];
  var puluhan = ['', 'Sepuluh', 'Dua Puluh', 'Tiga Puluh', 'Empat Puluh', 'Lima Puluh',
                 'Enam Puluh', 'Tujuh Puluh', 'Delapan Puluh', 'Sembilan Puluh'];

  if (num < 10) return satuan[num];
  if (num === 10) return 'Sepuluh';
  if (num < 20) {
    if (num === 11) return 'Sebelas';
    return satuan[num - 10] + ' Belas';
  }
  if (num < 100) {
    var puluh = Math.floor(num / 10);
    var satu = num % 10;
    if (satu === 0) return puluhan[puluh];
    return puluhan[puluh] + ' ' + satuan[satu];
  }
  if (num < 200) {
    if (num === 100) return 'Seratus';
    return 'Seratus ' + angkaTerbilang(num - 100);
  }
  if (num < 1000) {
    var ratus = Math.floor(num / 100);
    var sisaRatus = num % 100;
    if (sisaRatus === 0) return satuan[ratus] + ' Ratus';
    return satuan[ratus] + ' Ratus ' + angkaTerbilang(sisaRatus);
  }
  if (num < 2000) {
    var sisaRibu = num - 1000;
    if (sisaRibu === 0) return 'Seribu';
    return 'Seribu ' + angkaTerbilang(sisaRibu);
  }
  if (num < 1000000) {
    var ribu = Math.floor(num / 1000);
    var sisaRibu2 = num % 1000;
    if (sisaRibu2 === 0) return angkaTerbilang(ribu) + ' Ribu';
    return angkaTerbilang(ribu) + ' Ribu ' + angkaTerbilang(sisaRibu2);
  }
  return num.toString();
}

/**
 * Format bytes → string (B / KB / MB)
 */
function formatBytesPKKS_(bytes) {
  if (bytes < 1024) return bytes + ' B';
  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
  return (bytes / (1024 * 1024)).toFixed(2) + ' MB';
}

/**
 * Potong angka ke 2 desimal TANPA pembulatan, selalu 2 digit.
 * 90       → "90.00"
 * 90.5     → "90.50"
 * 90.256   → "90.25"
 * 90.987   → "90.98"
 * 88.125   → "88.12"
 */
function potongDuaDesimal(angka) {
  if (angka === null || angka === undefined || angka === '') return '';
  var num = parseFloat(angka);
  if (isNaN(num)) return '';

  var s = String(num);
  var match = s.match(/^(-?\d+)(?:\.(\d+))?/);
  if (!match) return '';

  var intPart = match[1];
  var decPart = match[2] || '';

  decPart = decPart.substring(0, 2);
  while (decPart.length < 2) decPart += '0';

  return intPart + '.' + decPart;
}

function readUsersSheet(ss) {
  var sheet = ss.getSheetByName(SHEET_NAMES.pengguna);
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  var list = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[0]) continue;
    list.push({
      pengguna: String(row[0]),
      kata_sandi: String(row[1]),
      peran: row[2] || 'admin',
      nama_lengkap: row[3] || row[0]
    });
  }
  return list;
}

function handleUserLogin(ss, username, password) {
  try {
    var sheet = ss.getSheetByName(SHEET_NAMES.pengguna);
    if (!sheet) return { status: 'error', message: 'Sheet Users tidak ditemukan!' };

    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return { status: 'error', message: 'Tidak ada user terdaftar!' };

    var u = String(username || '').trim();
    var p = String(password || '').trim();

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var ru = String(row[USER_COL.USERNAME] || '').trim();
      var rp = String(row[USER_COL.PASSWORD] || '').trim();
      if (ru === u && rp === p) {
        return {
          status: 'success',
          user: {
            username: ru,
            nama: String(row[USER_COL.NAMA_USER] || '').trim() || ru,
            role: String(row[USER_COL.ROLE] || 'user').trim().toLowerCase()
          }
        };
      }
    }
    return { status: 'error', message: 'Akses anda telah dihapus!' };
  } catch (e) {
    console.error('Error handleUserLogin:', e);
    return { status: 'error', message: e.toString() };
  }
}

  } catch (e) {
    return { status: 'error', message: e.toString() };

function updateSiswaDirect(siswa, noIndukLama) {
  try {
    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.siswa);
    if (!sheet) return { status: 'error', message: 'Sheet DataSiswa tidak ditemukan!' };

    var data = sheet.getDataRange().getValues();
    var cari = String(noIndukLama || siswa.noInduk || '').trim();
    var baris = -1;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][SISWA_COL.NO_INDUK] || '').trim() === cari) {
        baris = i + 1;
        break;
      }

    if (baris === -1) {
      return { status: 'error', message: 'Siswa No Induk ' + cari + ' tidak ditemukan di sheet!' };
    }

    var row = buildSiswaRow_(siswa);
    sheet.getRange(baris, 1, 1, row.length).setValues([row]);
    return { status: 'success', message: 'Siswa ' + siswa.nama + ' berhasil diperbarui.' };
  } catch (e) {
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Bulk update 1 field pada siswa / nabil / sorogan
 * params = { type, newValue, noIndukList }
 */
function bulkUpdateField(params) {
  try {
    var ss = getSpreadsheet();
    var type = params.type;
    var newValue = params.newValue;
    var noIndukList = params.noIndukList || [];

    if (!type || !newValue || noIndukList.length === 0) {
      return { success: false, message: 'Parameter tidak lengkap!' };
    }

    var sheet = null, colIndex = -1, noIndukCol = 0;

    if (type === 'kelas') {
      sheet = ss.getSheetByName(SHEET_NAMES.siswa);
      colIndex = SISWA_COL.KELAS;
      noIndukCol = SISWA_COL.NO_INDUK;
    } else if (type === 'nabil') {
      sheet = ss.getSheetByName(SHEET_NAMES.nabil);
      colIndex = NABIL_COL.KELOMPOK_NABIL;
      noIndukCol = NABIL_COL.NO_INDUK;
    } else if (type === 'sorogan') {
      sheet = ss.getSheetByName(SHEET_NAMES.sorogan);
      colIndex = SOROGAN_COL.PEMBIMBING;
      noIndukCol = SOROGAN_COL.NO_INDUK;
    } else {
      return { success: false, message: 'Tipe tidak dikenali: ' + type };
    }

    if (!sheet) return { success: false, message: 'Sheet tidak ditemukan!' };

    var data = sheet.getDataRange().getValues();
    var set = {};
    noIndukList.forEach(function(ni) { set[String(ni).trim()] = true; });

    var updatedCount = 0;
    for (var i = 1; i < data.length; i++) {
      var noInduk = String(data[i][noIndukCol] || '').trim();
      if (noInduk && set[noInduk]) {
        sheet.getRange(i + 1, colIndex + 1).setValue(newValue);
        updatedCount++;
      }
    }

    SpreadsheetApp.flush();
    return {
      success: true,
      message: 'Berhasil mengubah ' + updatedCount + ' baris',
      updated: updatedCount
    };
  } catch (e) {
    console.error('Error bulkUpdateField:', e);
    return { success: false, message: e.toString() };
  }
}

/**
 * Tambah siswa ke Nabil / Sorogan
 * params = { type: 'nabil'|'sorogan', newValue, noIndukList }
 */
function addSiswaToEkstra(params) {
  try {
    var ss = getSpreadsheet();
    var type = params.type;
    var newValue = params.newValue;
    var noIndukList = params.noIndukList || [];

    if (!type || !newValue || noIndukList.length === 0) {
      return { success: false, message: 'Parameter tidak lengkap!' };
    }

    var sheetName = type === 'nabil' ? SHEET_NAMES.nabil : SHEET_NAMES.sorogan;
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) return { success: false, message: 'Sheet ' + sheetName + ' tidak ditemukan!' };

    // Peta siswa dari DataSiswa
    var siswaSheet = ss.getSheetByName(SHEET_NAMES.siswa);
    var siswaData = siswaSheet.getDataRange().getValues();
    var siswaMap = {};
    for (var i = 1; i < siswaData.length; i++) {
      var ni = String(siswaData[i][SISWA_COL.NO_INDUK] || '').trim();
      if (ni) {
        siswaMap[ni] = {
          noInduk: ni,
          nama:    String(siswaData[i][SISWA_COL.NAMA]   || '').trim(),
          alamat:  String(siswaData[i][SISWA_COL.ALAMAT] || '').trim(),
          kelas:   String(siswaData[i][SISWA_COL.KELAS]  || '').trim()
        };
      }
    }

    // Yang sudah ada di sheet ekstra
    var existingData = sheet.getDataRange().getValues();
    var existingSet = {};
    for (var j = 1; j < existingData.length; j++) {
      var ni2 = String(existingData[j][0] || '').trim();
      if (ni2) existingSet[ni2] = true;
    }

    var added = 0, skipped = 0;
    for (var k = 0; k < noIndukList.length; k++) {
      var noInduk = String(noIndukList[k]).trim();
      if (existingSet[noInduk]) { skipped++; continue; }

      var siswa = siswaMap[noInduk];
      if (!siswa) continue;

      sheet.appendRow([siswa.noInduk, siswa.nama, siswa.alamat, siswa.kelas, newValue]);
      added++;
    }

    SpreadsheetApp.flush();
    return {
      success: true,
      message: 'Berhasil menambahkan ' + added + ' siswa ke ' + (type === 'nabil' ? 'Nabil' : 'Sorogan') + ' (dilewati: ' + skipped + ')',
      added: added, skipped: skipped
    };
  } catch (e) {
    return { success: false, message: e.toString() };
  }
}

/**
 * Baca ringkasan nilai (rata-rata per semester + avg)
 */
function readNilaiSheet(ss) {
  var sheet = ss.getSheetByName(SHEET_NAMES.nilai);
  if (!sheet) return [];

  var dataValues  = sheet.getDataRange().getValues();
  if (dataValues.length <= 1) return [];

  var dataDisplay = sheet.getDataRange().getDisplayValues();

  var RATA_COLS = {
    smt1: getColumnIndex(NILAI_COL.KLS1_SMT1.rataRata),
    smt2: getColumnIndex(NILAI_COL.KLS1_SMT2.rataRata),
    smt3: getColumnIndex(NILAI_COL.KLS2_SMT1.rataRata),
    smt4: getColumnIndex(NILAI_COL.KLS2_SMT2.rataRata),
    smt5: getColumnIndex(NILAI_COL.KLS3_SMT1.rataRata),
    smt6: getColumnIndex(NILAI_COL.KLS3_SMT2.rataRata)
  };

  function formatNilai(str) {
    if (str === null || str === undefined) return '';
    var s = String(str).trim();
    if (s === '' || s === '-') return '';

    var isNumber = /^-?\d+(\.\d+)?$/.test(s);
    if (!isNumber) return s;

    var parts = s.split('.');
    if (parts.length === 1) return parts[0];

    return parts[0] + '.' + parts[1].substring(0, 2);
  }

  function toNum(str) {
    if (str === '' || str === null || str === undefined) return 0;
    var num = parseFloat(str);
    return isNaN(num) ? 0 : num;
  }

  var list = [];
  for (var i = 1; i < dataValues.length; i++) {
    var row    = dataValues[i];
    var rowDisp = dataDisplay[i];

    var noInduk = String(row[0] || '').trim();
    var nama    = String(row[1] || '').trim();
    if (!noInduk && !nama) continue;

    if (nama.toLowerCase().indexOf('angkatan') !== -1) continue;
    if (noInduk.toLowerCase().indexOf('angkatan') !== -1) continue;

    var smt1Str = formatNilai(rowDisp[RATA_COLS.smt1]);
    var smt2Str = formatNilai(rowDisp[RATA_COLS.smt2]);
    var smt3Str = formatNilai(rowDisp[RATA_COLS.smt3]);
    var smt4Str = formatNilai(rowDisp[RATA_COLS.smt4]);
    var smt5Str = formatNilai(rowDisp[RATA_COLS.smt5]);
    var smt6Str = formatNilai(rowDisp[RATA_COLS.smt6]);

    var smt1 = toNum(smt1Str), smt2 = toNum(smt2Str), smt3 = toNum(smt3Str);
    var smt4 = toNum(smt4Str), smt5 = toNum(smt5Str), smt6 = toNum(smt6Str);

    var total = smt1 + smt2 + smt3 + smt4 + smt5 + smt6;
    var count = 0;
    if (smt1 > 0) count++;
    if (smt2 > 0) count++;
    if (smt3 > 0) count++;
    if (smt4 > 0) count++;
    if (smt5 > 0) count++;
    if (smt6 > 0) count++;
    var avg = count > 0 ? (total / count) : 0;

    list.push({
      noInduk: noInduk, nama: nama,
      smt1: smt1Str, smt2: smt2Str, smt3: smt3Str,

/**
 * Ambil detail nilai lengkap per mata pelajaran (untuk UI & cetak)
 */
function getDetailNilaiData(noInduk) {
  try {
    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.nilai);
    if (!sheet) return null;

    var lastCol = sheet.getLastColumn();
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return null;

    var rowIndex = -1;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0] || '').trim() === String(noInduk).trim()) {
        rowIndex = i;
        break;
      }
    }
    if (rowIndex === -1) return null;

    var rowDisplay = sheet.getRange(rowIndex + 1, 1, 1, lastCol).getDisplayValues()[0];
    var rowData = data[rowIndex];

    function formatNilai(str) {
      if (str === null || str === undefined) return null;
      var s = String(str).trim();
      if (s === '' || s === '-') return null;
      var isNumber = /^-?\d+(\.\d+)?$/.test(s);
      if (!isNumber) return s;
      var parts = s.split('.');
      if (parts.length === 1) return parts[0];
      return parts[0] + '.' + parts[1].substring(0, 2);
    }

    var result = {
      noInduk: String(rowData[0] || '').trim(),
      nama:    String(rowData[1] || '').trim() || 'Santri',
      semesters: [],
      tahun: {
        kls1smt1: rowDisplay[getColumnIndex(NILAI_COL.TAHUN.kls1smt1)] || '',
        kls1smt2: rowDisplay[getColumnIndex(NILAI_COL.TAHUN.kls1smt2)] || '',
        kls2smt1: rowDisplay[getColumnIndex(NILAI_COL.TAHUN.kls2smt1)] || '',
        kls2smt2: rowDisplay[getColumnIndex(NILAI_COL.TAHUN.kls2smt2)] || '',
        kls3smt1: rowDisplay[getColumnIndex(NILAI_COL.TAHUN.kls3smt1)] || '',
        kls3smt2: rowDisplay[getColumnIndex(NILAI_COL.TAHUN.kls3smt2)] || ''
      },
      dkn: [],
      rataRataIjazah: ''
    };

    var semesterConfigs = [
      { id: 'smt1', label: 'Semester 1 (Kelas X - Ganjil)',   tahunKey: 'kls1smt1', cfg: NILAI_COL.KLS1_SMT1 },
      { id: 'smt2', label: 'Semester 2 (Kelas X - Genap)',    tahunKey: 'kls1smt2', cfg: NILAI_COL.KLS1_SMT2 },
      { id: 'smt3', label: 'Semester 3 (Kelas XI - Ganjil)',  tahunKey: 'kls2smt1', cfg: NILAI_COL.KLS2_SMT1 },
      { id: 'smt4', label: 'Semester 4 (Kelas XI - Genap)',   tahunKey: 'kls2smt2', cfg: NILAI_COL.KLS2_SMT2 },
      { id: 'smt5', label: 'Semester 5 (Kelas XII - Ganjil)', tahunKey: 'kls3smt1', cfg: NILAI_COL.KLS3_SMT1 },
      { id: 'smt6', label: 'Semester 6 (Kelas XII - Genap)',  tahunKey: 'kls3smt2', cfg: NILAI_COL.KLS3_SMT2 }
    ];

    var mapelList = MAPEL_NILAI || [];

    // ===== DKN: rata-rata teori per mapel dari 6 semester =====
    var nilaiPerMapel = {};
    semesterConfigs.forEach(function(c) {
      var mapelAwal   = getColumnIndex(c.cfg.mapelAwal);
      var praktekAwal = getColumnIndex(c.cfg.praktekAwal);

      for (var m = 0; m < mapelList.length; m++) {
        if (!nilaiPerMapel[m]) nilaiPerMapel[m] = { teori: [], praktek: [] };

        var tv = parseFloat(String(rowData[mapelAwal + m]).replace(',', '.'));
        var pv = parseFloat(String(rowData[praktekAwal + m]).replace(',', '.'));

        if (!isNaN(tv)) nilaiPerMapel[m].teori.push(tv);
        if (!isNaN(pv)) nilaiPerMapel[m].praktek.push(pv);
      }
    });

    for (var m = 0; m < mapelList.length; m++) {
      var mp = nilaiPerMapel[m] || { teori: [], praktek: [] };
      var semua = mp.teori;

      var avg = 0;
      if (semua.length > 0) {
        var total = 0;
        for (var k = 0; k < semua.length; k++) total += semua[k];
        avg = total / semua.length;
      }

      result.dkn.push({
        nama:  mapelList[m] || ('DKN ' + (m + 1)),
        nilai: (semua.length === 0) ? null : potongDuaDesimal(avg)
      });
    }

    // ===== Rata-rata Ijazah =====
    var totalDkn = 0, countDkn = 0;
    result.dkn.forEach(function(item) {
      var num = parseFloat(item.nilai);
      if (!isNaN(num)) { totalDkn += num; countDkn++; }
    });
    result.rataRataIjazah = countDkn > 0 ? potongDuaDesimal(totalDkn / countDkn) : '';

    // ===== Detail per Semester =====
    semesterConfigs.forEach(function(c) {
      var mapelAwal   = getColumnIndex(c.cfg.mapelAwal);
      var mapelAkhir  = getColumnIndex(c.cfg.mapelAkhir);
      var praktekAwal = getColumnIndex(c.cfg.praktekAwal);
      var rataRataIdx = getColumnIndex(c.cfg.rataRata);

      var sem = {
        id: c.id,
        label: c.label,
        tahun: result.tahun[c.tahunKey] || '',
        rataRata: formatNilai(rowDisplay[rataRataIdx]),
        jumlah:   formatNilai(rowDisplay[getColumnIndex(c.cfg.jumlah)]),
        mapel: []
      };

      var teoriCount = mapelAkhir - mapelAwal + 1;
      for (var i = 0; i < teoriCount && i < mapelList.length; i++) {
        sem.mapel.push({
          mapel:   mapelList[i] || ('Mata Pelajaran ' + (i + 1)),
          teori:   formatNilai(rowDisplay[mapelAwal + i]),
          praktek: formatNilai(rowDisplay[praktekAwal + i])
        });
      }
      result.semesters.push(sem);
    });

    return result;
  } catch (e) {
    console.error('Error getDetailNilaiData:', e);
    return null;
  }
}

/**
 * Simpan detail nilai per semester + DKN
 */
function saveDetailNilaiDirect(payload) {
  try {
    if (!payload || !payload.noInduk) {
      return { status: 'error', message: 'No Induk tidak valid!' };
    }

    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.nilai);
    if (!sheet) return { status: 'error', message: 'Sheet DataNilai tidak ditemukan!' };

    var data = sheet.getDataRange().getValues();
    var baris = -1;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0] || '').trim() === String(payload.noInduk).trim()) {
        baris = i + 1;
        break;
      }
    }
    if (baris === -1) return { status: 'error', message: 'Siswa No Induk ' + payload.noInduk + ' tidak ditemukan!' };

    var lastCol = sheet.getLastColumn();
    var row = sheet.getRange(baris, 1, 1, lastCol).getValues()[0];

    // Pastikan row cukup panjang (minimal 300 kolom)
    while (row.length < 300) row.push('');

    var semesterConfigs = {
      smt1: NILAI_COL.KLS1_SMT1,
      smt2: NILAI_COL.KLS1_SMT2,
      smt3: NILAI_COL.KLS2_SMT1,
      smt4: NILAI_COL.KLS2_SMT2,
      smt5: NILAI_COL.KLS3_SMT1,
      smt6: NILAI_COL.KLS3_SMT2
    };

    Object.keys(semesterConfigs).forEach(function(smtId) {
      var cfg = semesterConfigs[smtId];
      var nilaiSmt = (payload.semesters && payload.semesters[smtId]) || [];

      var mapelAwal   = getColumnIndex(cfg.mapelAwal);
      var praktekAwal = getColumnIndex(cfg.praktekAwal);
      var jumlahIdx   = getColumnIndex(cfg.jumlah);
      var rataRataIdx = getColumnIndex(cfg.rataRata);

      var totalTeori = 0;
      var countTeori = 0;

      for (var m = 0; m < nilaiSmt.length; m++) {
        var item = nilaiSmt[m];
        if (!item) continue;

        var teoriVal   = (item.teori === null || item.teori === undefined || item.teori === '') ? '' : parseFloat(item.teori);
        var praktekVal = (item.praktek === null || item.praktek === undefined || item.praktek === '') ? '' : parseFloat(item.praktek);

        if (!isNaN(teoriVal)) {
          row[mapelAwal + m] = teoriVal;
          totalTeori += teoriVal;
          countTeori++;
        }
        if (!isNaN(praktekVal)) row[praktekAwal + m] = praktekVal;
      }

      if (nilaiSmt._jumlah !== undefined && nilaiSmt._jumlah !== null) {
        row[jumlahIdx] = Math.round(parseFloat(nilaiSmt._jumlah));
      } else {
        row[jumlahIdx] = countTeori > 0 ? Math.round(totalTeori) : '';
      }

      if (nilaiSmt._rataRata !== undefined && nilaiSmt._rataRata !== null) {
        row[rataRataIdx] = potongDuaDesimal(parseFloat(nilaiSmt._rataRata));
      } else {
        row[rataRataIdx] = countTeori > 0 ? potongDuaDesimal(totalTeori / countTeori) : '';
      }
    });

    // DKN (18 kolom)
    var dknAwal = getColumnIndex(NILAI_COL.IJAZAH.dknAwal);
    var dknArr = payload.dkn || [];
    var totalDkn = 0, countDkn = 0;

    for (var m = 0; m < dknArr.length; m++) {
      var v = dknArr[m];
      var num = (v === null || v === undefined || v === '') ? '' : parseFloat(v);
      if (!isNaN(num)) {
        row[dknAwal + m] = potongDuaDesimal(num);
        totalDkn += num;
        countDkn++;
      }
    }

    // Rata-rata Ijazah
    var rrIdx = getColumnIndex(NILAI_COL.RATA_RATAI_JAZAH);
    if (payload.rataRataIjazah !== undefined && payload.rataRataIjazah !== null && payload.rataRataIjazah !== '') {
      row[rrIdx] = potongDuaDesimal(parseFloat(payload.rataRataIjazah));
    } else {
      row[rrIdx] = countDkn > 0 ? potongDuaDesimal(totalDkn / countDkn) : '';
    }

    sheet.getRange(baris, 1, 1, row.length).setValues([row]);

    return { status: 'success', message: 'Data nilai berhasil diperbarui.' };
  } catch (e) {
    console.error('Error saveDetailNilaiDirect:', e);
    return { status: 'error', message: e.toString() };
  }
}

      return { status: 'error', message: 'Data jadwal tidak valid.' };
    }

    var ss = getSpreadsheet();

function uploadFilePKKS(payload) {
  try {
    if (!payload || !payload.base64 || !payload.nama) {
      return { status: 'error', message: 'Data tidak lengkap!' };
    }

    var ss = getSpreadsheet();
    var sheet = ensureSheetPKKS_(ss);

    // Tentukan folder tujuan
    var folderTujuan;
    if (payload.folderId) {
      try {
        folderTujuan = DriveApp.getFolderById(payload.folderId);
      } catch (e) {
        return { status: 'error', message: 'Folder tujuan tidak ditemukan!' };
      }
    } else {
      folderTujuan = DriveApp.getFolderById(FOLDER_PKKS_ID);
    }

    var bytes = Utilities.base64Decode(payload.base64);
    var blob = Utilities.newBlob(bytes, payload.mimeType || 'application/octet-stream', payload.nama);

    var file = folderTujuan.createFile(blob);
    var fileId = file.getId();
    var fileUrl = file.getUrl();
    var downloadUrl = 'https://drive.google.com/uc?export=download&id=' + fileId;

    var sizeStr = formatBytesPKKS_(blob.getBytes().length);
    var tanggal = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm');
    var no = sheet.getLastRow();

    sheet.appendRow([
      no,
      payload.nama,
      fileId,
      tanggal,
      sizeStr,
      payload.jenis || 'Lainnya',
      payload.keterangan || '',
      payload.uploader || 'admin'
    ]);

    return {
      status: 'success',
      message: 'File berhasil diupload!',
      fileId: fileId,
      fileUrl: fileUrl,
      downloadUrl: downloadUrl,
      nama: payload.nama
    };
  } catch (e) {
    console.error('Error uploadFilePKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

function hapusPKKS(fileId, rowIndex) {
  try {
    if (!fileId) return { status: 'error', message: 'File ID kosong!' };

    var ss = getSpreadsheet();
    var sheet = ensureSheetPKKS_(ss);

    try {
      DriveApp.getFileById(fileId).setTrashed(true);
    } catch (driveErr) {
      console.warn('File tidak ditemukan di Drive:', driveErr);
    }

    if (rowIndex && rowIndex > 1) {
      sheet.deleteRow(rowIndex);
    } else {
      var data = sheet.getDataRange().getValues();
      for (var i = data.length - 1; i >= 1; i--) {
        if (String(data[i][2] || '').trim() === String(fileId).trim()) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
    }

    renumberNoPKKS_(sheet);
    return { status: 'success', message: 'File berhasil dihapus.' };
  } catch (e) {
    console.error('Error hapusPKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

function renumberNoPKKS_(sheet) {
  try {
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) return;
    var range = sheet.getRange(2, 1, lastRow - 1, 1);
    var values = range.getValues();
    for (var i = 0; i < values.length; i++) values[i][0] = i + 1;
    range.setValues(values);
  } catch (e) {
    console.warn('renumberNoPKKS_ error:', e);
  }
}

/**
 * Ambil daftar subfolder di dalam folder PKKS
 */
function getDaftarFolderPKKS() {
  try {
    var folder = DriveApp.getFolderById(FOLDER_PKKS_ID);
    var folders = folder.getFolders();
    var list = [];

    while (folders.hasNext()) {
      var f = folders.next();
      list.push({
        id: f.getId(),
        nama: f.getName(),
        url: f.getUrl(),
        tanggal: Utilities.formatDate(f.getDateCreated(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm')
      });
    }

    list.sort(function(a, b) {
      return a.nama.localeCompare(b.nama, 'id', { sensitivity: 'base' });
    });

    return { status: 'success', data: list };
  } catch (e) {
    console.error('Error getDaftarFolderPKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Buat subfolder baru di dalam folder PKKS (atau di parent tertentu)
 */
function buatFolderPKKS(namaFolder, parentFolderId) {
  try {
    if (!namaFolder || String(namaFolder).trim() === '') {
      return { status: 'error', message: 'Nama folder tidak boleh kosong!' };
    }

    var nama = String(namaFolder).trim();

    var parent;
    if (parentFolderId) {
      try {
        parent = DriveApp.getFolderById(parentFolderId);
      } catch (e) {
        return { status: 'error', message: 'Folder induk tidak ditemukan!' };
      }
    } else {
      parent = DriveApp.getFolderById(FOLDER_PKKS_ID);
    }

    // Cek duplikat nama
    var existing = parent.getFoldersByName(nama);
    if (existing.hasNext()) {
      var f = existing.next();
      return {
        status: 'success',
        message: 'Folder "' + nama + '" sudah ada, menggunakan folder ini.',
        data: { id: f.getId(), nama: f.getName(), url: f.getUrl() },
        existing: true
      };
    }

    var newFolder = parent.createFolder(nama);
    return {
      status: 'success',
      message: 'Folder "' + nama + '" berhasil dibuat.',
      data: { id: newFolder.getId(), nama: newFolder.getName(), url: newFolder.getUrl() },
      existing: false
    };
  } catch (e) {
    console.error('Error buatFolderPKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Hapus subfolder PKKS (pindah ke trash + hapus baris sheet)
 */
function hapusFolderPKKS(folderId) {
  try {
    if (!folderId) return { status: 'error', message: 'Folder ID tidak valid!' };

    var folder = DriveApp.getFolderById(folderId);

    // 1. Kumpulkan semua fileId di dalam folder (rekursif)
    var fileIds = [];
    collectFileIds_(folder, fileIds);

    // 2. Hapus baris sheet untuk setiap fileId
    if (fileIds.length > 0) {
      var ss = getSpreadsheet();
      var sheet = ensureSheetPKKS_(ss);
      var data = sheet.getDataRange().getValues();

      var setIds = {};
      fileIds.forEach(function(id) { setIds[id] = true; });

      for (var i = data.length - 1; i >= 1; i--) {
        var fid = String(data[i][2] || '').trim();
        if (setIds[fid]) {
          sheet.deleteRow(i + 1);
        }
      }

      renumberNoPKKS_(sheet);
    }

    // 3. Pindah folder ke trash
    folder.setTrashed(true);

    return {
      status: 'success',
      message: 'Folder berhasil dihapus (' + fileIds.length + ' file di dalamnya).'
    };
  } catch (e) {
    console.error('Error hapusFolderPKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Ganti nama folder PKKS
 */
function renameFolderPKKS(folderId, namaBaru) {
  try {
    if (!folderId) return { status: 'error', message: 'Folder ID tidak valid!' };
    if (!namaBaru || String(namaBaru).trim() === '') {
      return { status: 'error', message: 'Nama folder tidak boleh kosong!' };
    }

    var nama = String(namaBaru).trim();
    var folder = DriveApp.getFolderById(folderId);

    // Cek duplikat di parent yang sama
    var parents = folder.getParents();
    if (parents.hasNext()) {
      var parent = parents.next();
      var existing = parent.getFoldersByName(nama);
      while (existing.hasNext()) {
        var f = existing.next();
        if (f.getId() !== folderId) {
          return { status: 'error', message: 'Folder dengan nama "' + nama + '" sudah ada!' };
        }
      }
    }

    var namaLama = folder.getName();
    folder.setName(nama);

    return {
      status: 'success',
      message: 'Folder "' + namaLama + '" berhasil diubah menjadi "' + nama + '".',
      data: { id: folderId, nama: nama }
    };
  } catch (e) {
    console.error('Error renameFolderPKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Hapus file PKKS hanya berdasarkan fileId
 */
function hapusFilePKKSById(fileId) {
  try {
    if (!fileId) return { status: 'error', message: 'File ID kosong!' };

    var ss = getSpreadsheet();
    var sheet = ensureSheetPKKS_(ss);

    try {
      DriveApp.getFileById(fileId).setTrashed(true);
    } catch (driveErr) {
      console.warn('File tidak ditemukan di Drive:', driveErr);
    }

    var data = sheet.getDataRange().getValues();
    for (var i = data.length - 1; i >= 1; i--) {
      if (String(data[i][2] || '').trim() === String(fileId).trim()) {
        sheet.deleteRow(i + 1);
        break;
      }
    }

    renumberNoPKKS_(sheet);
    return { status: 'success', message: 'File berhasil dihapus.' };
  } catch (e) {
    console.error('Error hapusFilePKKSById:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Pindah file PKKS dari folder lama ke folder baru
 */
function pindahFilePKKS(fileId, targetFolderId) {
  try {
    if (!fileId) return { status: 'error', message: 'File ID tidak valid!' };

    var file;
    try {
      file = DriveApp.getFileById(fileId);
    } catch (e) {
      return { status: 'error', message: 'File tidak ditemukan di Drive!' };
    }

    var targetFolder;
    if (targetFolderId) {
      try {
        targetFolder = DriveApp.getFolderById(targetFolderId);
      } catch (e) {
        return { status: 'error', message: 'Folder tujuan tidak ditemukan!' };
      }
    } else {
      targetFolder = DriveApp.getFolderById(FOLDER_PKKS_ID);
    }

    // Hapus dari semua parent folder saat ini
    var parents = file.getParents();
    while (parents.hasNext()) {
      var parent = parents.next();
      parent.removeFile(file);
    }

    // Tambah ke folder tujuan
    targetFolder.addFile(file);

    return {
      status: 'success',
      message: 'File "' + file.getName() + '" berhasil dipindah ke folder "' + targetFolder.getName() + '".'
    };
  } catch (e) {
    console.error('Error pindahFilePKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Helper: kumpulkan semua fileId di dalam folder (rekursif)
 */
function collectFileIds_(folder, arr) {
  var files = folder.getFiles();
  while (files.hasNext()) {
    arr.push(files.next().getId());
  }
  var subs = folder.getFolders();
  while (subs.hasNext()) {
    collectFileIds_(subs.next(), arr);
  }
}

/**
 * Ambil daftar file di dalam subfolder PKKS
 */
function getFileDalamFolderPKKS(folderId) {
  try {
    if (!folderId) return { status: 'error', message: 'Folder ID tidak valid!' };

    var folder = DriveApp.getFolderById(folderId);
    var files = folder.getFiles();
    var list = [];

    while (files.hasNext()) {
      var f = files.next();
      var sizeBytes = f.getSize();
      list.push({
        id: f.getId(),
        nama: f.getName(),
        url: f.getUrl(),
        downloadUrl: 'https://drive.google.com/uc?export=download&id=' + f.getId(),
        mimeType: f.getMimeType(),
        ukuran: formatBytesPKKS_(sizeBytes),
        tanggal: Utilities.formatDate(f.getDateCreated(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm')
      });
    }

    list.sort(function(a, b) {
      return b.tanggal.localeCompare(a.tanggal);
    });

    var folderInfo = {
      id: folder.getId(),
      nama: folder.getName(),
      url: folder.getUrl()
    };

    return { status: 'success', data: list, folder: folderInfo };
  } catch (e) {
    console.error('Error getFileDalamFolderPKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Ambil isi folder PKKS: subfolder + file.
 * Menggantikan getFileDalamFolderPKKS dengan struktur lengkap.
 */
function getIsiFolderPKKS(folderId) {
  try {
    if (!folderId) return { status: 'error', message: 'Folder ID tidak valid!' };

    var folder = DriveApp.getFolderById(folderId);

    // ===== Ambil subfolder =====
    var subfolders = [];
    var foldersIter = folder.getFolders();
    while (foldersIter.hasNext()) {
      var f = foldersIter.next();
      subfolders.push({
        id: f.getId(),
        nama: f.getName(),
        url: f.getUrl(),
        tanggal: Utilities.formatDate(f.getDateCreated(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm')
      });
    }
    subfolders.sort(function(a, b) {
      return a.nama.localeCompare(b.nama, 'id', { sensitivity: 'base' });
    });

    // ===== Ambil file =====
    var files = [];
    var filesIter = folder.getFiles();
    while (filesIter.hasNext()) {
      var file = filesIter.next();
      var sizeBytes = file.getSize();
      files.push({
        id: file.getId(),
        nama: file.getName(),
        url: file.getUrl(),
        downloadUrl: 'https://drive.google.com/uc?export=download&id=' + file.getId(),
        mimeType: file.getMimeType(),
        ukuran: formatBytesPKKS_(sizeBytes),
        tanggal: Utilities.formatDate(file.getDateCreated(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm')
      });
    }
    files.sort(function(a, b) {
      return b.tanggal.localeCompare(a.tanggal);
    });

    // ===== Ambil info parent (untuk breadcrumb) =====
    var parentInfo = null;
    var parents = folder.getParents();
    if (parents.hasNext()) {
      var p = parents.next();
      // Cek apakah parent = root PKKS
      var isRoot = (p.getId() === FOLDER_PKKS_ID);
      parentInfo = {
        id: p.getId(),
        nama: isRoot ? 'PKKS' : p.getName(),
        isRoot: isRoot
      };
    }

    var folderInfo = {
      id: folder.getId(),
      nama: folder.getName(),
      url: folder.getUrl(),
      parent: parentInfo
    };

    return {
      status: 'success',
      folders: subfolders,
      files: files,
      folder: folderInfo
    };
  } catch (e) {
    console.error('Error getIsiFolderPKKS:', e);
    return { status: 'error', message: e.toString() };
  }
}

function getAbsenByKelas(kelas) {
  try {
    var siswaData = getSiswaData();
    var result = [];
    var target = String(kelas).trim().toLowerCase();

    for (var i = 0; i < siswaData.length; i++) {
      var s = siswaData[i];
      var kelasSiswa = (s.kelas || '').toLowerCase();
      var isMatch = false;

      if (kelasSiswa.trim() === target) isMatch = true;
      else if (kelasSiswa.includes('berhenti ' + target)) isMatch = true;
      else if (kelasSiswa.includes('berhenti' + target)) isMatch = true;
      else if (kelasSiswa.includes('berhenti') && kelasSiswa.includes(target)) isMatch = true;

      if (isMatch) {
        var isBerhenti = kelasSiswa.includes('berhenti');
        result.push({
          no: result.length + 1,
          noInduk: s.noInduk || '',
          nama:    s.nama || '',
          alamat:  s.alamat || '',
          berhenti: isBerhenti,
          status:  s.status || 'Aktif'
        });
      }
    }

    result.sort(function(a, b) {
      if (a.berhenti && !b.berhenti) return 1;
      if (!a.berhenti && b.berhenti) return -1;
      return a.nama.localeCompare(b.nama, 'id', { sensitivity: 'base' });
    });
    result.forEach(function(r, index) { r.no = index + 1; });

    return result;
  } catch (e) {
    console.error('Error getAbsenByKelas:', e);
    return [];
  }
}

function exportRaportToPDF(noInduk, semester, tanggal) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    var shSiswa = ss.getSheetByName(SHEET_NAMES.siswa);
    if (!shSiswa) throw new Error('Sheet DataSiswa tidak ditemukan!');

    var siswaData = shSiswa.getDataRange().getValues();
    var siswaInfo = null;
    for (var i = 1; i < siswaData.length; i++) {
      if (String(siswaData[i][SISWA_COL.NO_INDUK] || '').trim() === String(noInduk).trim()) {
        siswaInfo = {
          nama:    siswaData[i][SISWA_COL.NAMA] || '',
          noInduk: String(siswaData[i][SISWA_COL.NO_INDUK] || '').trim(),
          kelas:   siswaData[i][SISWA_COL.KELAS] || '',
          nisn:    siswaData[i][SISWA_COL.NISN] || '',
          tetala:  siswaData[i][SISWA_COL.TETALA] || ''
        };
        break;
      }
    }
    if (!siswaInfo) throw new Error('Data siswa tidak ditemukan!');

    var shNilai = ss.getSheetByName(SHEET_NAMES.nilai);
    if (!shNilai) throw new Error('Sheet DataNilai tidak ditemukan!');

    var nilaiData = shNilai.getDataRange().getValues();
    var nilaiRow = null;
    for (var j = 1; j < nilaiData.length; j++) {
      if (String(nilaiData[j][0] || '').trim() === String(noInduk).trim()) {
        nilaiRow = nilaiData[j];
        break;
      }
    }
    if (!nilaiRow) throw new Error('Data nilai tidak ditemukan!');

    var semesterMap = {
      1: { cfg: NILAI_COL.KLS1_SMT1, label: 'Semester 1' },
      2: { cfg: NILAI_COL.KLS1_SMT2, label: 'Semester 2' },
      3: { cfg: NILAI_COL.KLS2_SMT1, label: 'Semester 3' },
      4: { cfg: NILAI_COL.KLS2_SMT2, label: 'Semester 4' },
      5: { cfg: NILAI_COL.KLS3_SMT1, label: 'Semester 5' },
      6: { cfg: NILAI_COL.KLS3_SMT2, label: 'Semester 6' }
    };
    var sm = semesterMap[semester];
    if (!sm) throw new Error('Semester tidak valid!');

    var mapelAwal    = getColumnIndex(sm.cfg.mapelAwal);
    var mapelAkhir   = getColumnIndex(sm.cfg.mapelAkhir);
    var praktekAwal  = getColumnIndex(sm.cfg.praktekAwal);
    var praktekAkhir = getColumnIndex(sm.cfg.praktekAkhir);
    var tahunIdx     = getColumnIndex(sm.cfg.tahun);

    var mapelList = MAPEL_NILAI || [];
    var teoriData = [], praktekData = [];
    var totalTeori = 0, countTeori = 0;

    for (var k = mapelAwal; k <= mapelAkhir && k < nilaiRow.length; k++) {
      var idx = k - mapelAwal;
      if (idx < mapelList.length) {
        var val = parseFloat(nilaiRow[k]);
        var nilai = isNaN(val) ? null : val;
        teoriData.push({ nama: mapelList[idx], nilai: nilai });
        if (nilai !== null) { totalTeori += nilai; countTeori++; }
      }
    }
    for (var p = praktekAwal; p <= praktekAkhir && p < nilaiRow.length; p++) {
      var idxP = p - praktekAwal;
      if (idxP < mapelList.length) {
        var valP = parseFloat(nilaiRow[p]);
        praktekData.push({ nama: mapelList[idxP], nilai: isNaN(valP) ? null : valP });
      }
    }

    var rataTeori = countTeori > 0 ? totalTeori / countTeori : 0;
    var tahun = nilaiRow[tahunIdx] || '';
    var kelas = 1;
    if (semester >= 3 && semester <= 4) kelas = 2;
    else if (semester >= 5 && semester <= 6) kelas = 3;

    var templateData = {
      nama: siswaInfo.nama, noInduk: siswaInfo.noInduk,
      kelas: kelas, semester: semester, tahun: tahun,
      tanggal: tanggal || '',
      teori: teoriData, praktek: praktekData,
      jumlah: { nilai: totalTeori },
      rata: { nilai: rataTeori }
    };

    var hasil = isiTemplateRaport(templateData);
    if (!hasil.success) throw new Error(hasil.message || 'Gagal mengisi template');

    SpreadsheetApp.flush();
    Utilities.sleep(1500);

    var shRaport = ss.getSheetByName(SHEET_NAMES.laporan);
    if (!shRaport) throw new Error('Sheet RaportSma tidak ditemukan!');

    var gid = shRaport.getSheetId();
    var filename = noInduk + '_' + siswaInfo.nama + '_Raport_S' + semester + '.pdf';

    var result = exportPDF(gid, filename);
    if (!result.success) throw new Error(result.message || 'Gagal generate PDF');
    return result;
  } catch (e) {
    console.error('Error exportRaportToPDF:', e);
    return { success: false, message: e.toString() };
  }
}

function isiTemplateRaport(data) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var shRaport = ss.getSheetByName(SHEET_NAMES.laporan);
    var shMap    = ss.getSheetByName(SHEET_NAMES.mappingRaport);
    if (!shRaport) throw new Error('Sheet RaportSma tidak ditemukan!');
    if (!shMap) throw new Error('Sheet MappingRaport tidak ditemukan!');

    var mapping = shMap.getDataRange().getValues();
    var map = {};
    for (var i = 1; i < mapping.length; i++) {
      var key = String(mapping[i][0] || '').trim();
      if (!key) continue;
      map[key] = {
        teori:           String(mapping[i][1] || '').trim(),
        praktek:         String(mapping[i][2] || '').trim(),
        terbilangTeori:  String(mapping[i][3] || '').trim(),
        terbilangPraktek: String(mapping[i][4] || '').trim()
      };
    }

    function setCell(key, val) {
      if (map[key] && map[key].teori) {
        try { shRaport.getRange(map[key].teori).setValue(val); } catch(e) {}
      }
    }
    function setTerbilang(key, val, type) {
      var cell = type === 'teori' ? (map[key] && map[key].terbilangTeori) : (map[key] && map[key].terbilangPraktek);
      if (cell) {
        try { shRaport.getRange(cell).setValue(angkaTerbilang(val)); } catch(e) {}
      }
    }

    setCell('Nama', ': ' + data.nama);
    setCell('no induk', ': ' + data.noInduk);
    setCell('No Induk', ': ' + data.noInduk);

    if (map['Kelas/Semester']) {
      var romawi = ['', 'X', 'XI', 'XII'][data.kelas] || data.kelas;
      shRaport.getRange(map['Kelas/Semester'].teori).setValue(': ' + romawi + ' / ' + (data.semester === 1 ? 'I' : 'II'));
    }
    setCell('Tahun', ': ' + data.tahun);
    if (data.tanggal) {
      setCell('Tanggal', data.tanggal);
      setCell('TANGGAL', data.tanggal);
    }

    var mapelList = MAPEL_NILAI || [];
    data.teori.forEach(function(item, index) {
      var nama = mapelList[index] || item.nama;
      if (map[nama]) {
        if (map[nama].teori) {
          try {
            shRaport.getRange(map[nama].teori).setValue(item.nilai);
            if (item.nilai !== null && item.nilai !== '') setTerbilang(nama, item.nilai, 'teori');
          } catch(e) {}
        }
        if (data.praktek && data.praktek[index] && map[nama].praktek) {
          try {
            shRaport.getRange(map[nama].praktek).setValue(data.praktek[index].nilai);
            if (data.praktek[index].nilai !== null && data.praktek[index].nilai !== '') setTerbilang(nama, data.praktek[index].nilai, 'praktek');
          } catch(e) {}
        }
      }
    });

    setCell('Jumlah', data.jumlah.nilai);
    setCell('Rata-rata', data.rata.nilai);
    return { success: true };
  } catch (e) {
    console.error('Error isiTemplateRaport:', e);
    return { success: false, message: e.toString() };
  }
}

  try {
    var url = 'https://docs.google.com/spreadsheets/d/' + SPREADSHEET_ID +
      '/export?format=pdf' +
      '&gid=' + gid +
      '&size=f4' +
      '&portrait=true' +
      '&fitw=false' +
      '&scale=4' +
      '&sheetnames=false' +
      '&printtitle=false' +
      '&gridlines=false' +
      '&pagenumbers=false' +
      '&fzr=false' +
      '&top_margin=0.2' +
      '&bottom_margin=0.2' +
      '&left_margin=0.2' +
      '&right_margin=0.2' +
      '&horizontal_alignment=CENTER' +
      '&vertical_alignment=TOP' +
      '&printselection=true';

    var response = UrlFetchApp.fetch(url, {
      headers: { Authorization: 'Bearer ' + token },

    return {
      success: true,
      filename: filename,
      pdf: Utilities.base64Encode(blob.getBytes()),
      size: blob.getBytes().length
    };
  } catch (e) {
    console.error('Error exportPDF:', e);
    return { success: false, message: e.toString() };
  }

   ========================================================= */

function exportDatabaseExcel(params) {
  try {
    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.siswa);
    if (!sheet) return { success: false, message: 'Sheet DataSiswa tidak ditemukan!' };

    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return { success: false, message: 'Tidak ada data!' };

    var headers  = data[0];
    var allRows  = data.slice(1);
    var namaPengunduh    = params.namaPengunduh    || 'Administrator';
    var waktuPengunduhan = params.waktuPengunduhan || new Date().toLocaleString('id-ID');

    var filteredRows = [];
    var filterLabels = [];
    var usedIndex = {};

    var filtersKelas    = params.filtersKelas    || [];
    var filtersAngkatan = params.filtersAngkatan || [];
    if (!Array.isArray(filtersKelas))    filtersKelas = [];
    if (!Array.isArray(filtersAngkatan)) filtersAngkatan = [];
    filtersKelas    = filtersKelas.filter(function(f)    { return f && String(f).trim() !== ''; });
    filtersAngkatan = filtersAngkatan.filter(function(f) { return f && String(f).trim() !== ''; });

    // Backward-compat: kalau params.filters dikirim
    if (filtersKelas.length === 0 && filtersAngkatan.length === 0) {
      var filtersLama = params.filters || [];
      filtersLama.forEach(function(f) {
        var fs = String(f).trim();
        if (/^\d{4}$/.test(fs)) filtersAngkatan.push(fs);
        else filtersKelas.push(fs);
      });
    }

    if (filtersKelas.length === 0 && filtersAngkatan.length === 0) {
      filteredRows = allRows;
      filterLabels.push('Semua Data');
    } else {
      if (filtersKelas.length > 0) {
        filterLabels.push('Kelas: ' + filtersKelas.join(', '));
        var targetKelas = filtersKelas.map(function(k) { return k.toLowerCase(); });
        for (var i = 0; i < allRows.length; i++) {
          var kelasSiswa = String(allRows[i][SISWA_COL.KELAS] || '').trim().toLowerCase();
          for (var t = 0; t < targetKelas.length; t++) {
            var tk = targetKelas[t];
            if (kelasSiswa === tk || kelasSiswa === ('berhenti ' + tk) || kelasSiswa.includes(tk)) {
              var key = 'r' + i;
              if (!usedIndex[key]) { usedIndex[key] = true; filteredRows.push(allRows[i]); }
              break;
            }
          }
        }
      }

      if (filtersAngkatan.length > 0) {
        filterLabels.push('Angkatan: ' + filtersAngkatan.join(', '));
        var targetAngkatan = filtersAngkatan.map(function(a) { return String(a).trim(); });
        for (var j = 0; j < allRows.length; j++) {
          var masukStr = String(allRows[j][SISWA_COL.MASUK] || '').trim();
          var m = masukStr.match(/\d{4}/);
          var tahunSiswa = m ? m[0] : '';
          if (targetAngkatan.indexOf(tahunSiswa) !== -1) {
            var keyJ = 'r' + j;
            if (!usedIndex[keyJ]) { usedIndex[keyJ] = true; filteredRows.push(allRows[j]); }
          }
        }
      }
    }

    if (filteredRows.length === 0) {
      return { success: false, message: 'Tidak ada data yang sesuai dengan filter!' };
    }

    // Bangun spreadsheet temporary
    var newSS = SpreadsheetApp.create('Database_Siswa_Temp_' + new Date().getTime());
    var newSheet = newSS.getActiveSheet();
    newSheet.setName('Database Siswa');

    newSheet.getRange(1, 1).setValue('Waktu Pengunduhan:');
    newSheet.getRange(1, 2).setValue(waktuPengunduhan);
    newSheet.getRange(2, 1).setValue('Nama Pengunduh:');
    newSheet.getRange(2, 2).setValue(namaPengunduh);
    newSheet.getRange(3, 1).setValue('Filter Pengunduhan:');
    newSheet.getRange(3, 2).setValue(filterLabels.join(', '));

    var startRow = 5;
    newSheet.getRange(startRow, 1, 1, headers.length).setValues([headers]);
    if (filteredRows.length > 0) {
      newSheet.getRange(startRow + 1, 1, filteredRows.length, headers.length).setValues(filteredRows);
    }

    newSheet.getRange(1, 1, 3, 1).setFontWeight('bold');
    newSheet.getRange(startRow, 1, 1, headers.length).setFontWeight('bold');
    newSheet.getRange(startRow, 1, 1, headers.length).setBackground('#e8f0fe');
    for (var c = 1; c <= headers.length; c++) newSheet.autoResizeColumn(c);
    newSheet.setFrozenRows(startRow);

    Utilities.sleep(2000);

    var tempId = newSS.getId();
    var url = 'https://docs.google.com/spreadsheets/d/' + tempId + '/export?format=xlsx';
    var token = ScriptApp.getOAuthToken();
    var response = UrlFetchApp.fetch(url, {
      headers: { Authorization: 'Bearer ' + token },
      muteHttpExceptions: true
    });

    try { DriveApp.getFileById(tempId).setTrashed(true); } catch(e) {}

    if (response.getResponseCode() !== 200) {
      return { success: false, message: 'Gagal generate Excel: ' + response.getResponseCode() };
    }

    var blob = response.getBlob();
    var tanggalFile = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd');
    var labelFile = filterLabels.length === 1
      ? filterLabels[0].replace(/[^a-zA-Z0-9]/g, '_')
      : 'Gabungan_' + filterLabels.length + '_Filter';
    var filename = 'Database_Siswa_' + labelFile + '_' + tanggalFile + '.xlsx';
    blob.setName(filename);

    return {
      success: true,
      excel: Utilities.base64Encode(blob.getBytes()),
      filename: filename,
      total: filteredRows.length,
      filters: filterLabels
    };
  } catch (e) {
    console.error('Error exportDatabaseExcel:', e);
    return { success: false, message: e.toString() };
  }
}

function readMappingRaportSheet(ss) {
  var sheet = ss.getSheetByName(SHEET_NAMES.mappingRaport);
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  var headers = data[0].map(function(h) { return String(h || '').trim().toUpperCase(); });

  var keyIdx = headers.indexOf('KEY');
  if (keyIdx === -1) keyIdx = headers.indexOf('KUNCI');
  if (keyIdx === -1) keyIdx = 0;

  var teoriIdx = headers.indexOf('CELL TEORI');
  if (teoriIdx === -1) teoriIdx = headers.indexOf('SEL TEORI');
  if (teoriIdx === -1) teoriIdx = 1;

  var praktekIdx = headers.indexOf('CELL PRAKTEK');
  if (praktekIdx === -1) praktekIdx = headers.indexOf('SEL PRAKTEK');
  if (praktekIdx === -1) praktekIdx = 2;

  var list = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var k = String(row[keyIdx] || '').trim();
    if (!k) continue;
    list.push({
      kunci:       k,
      cell_teori:  String(row[teoriIdx]   || '').trim().toUpperCase(),
      sel_praktik: String(row[praktekIdx] || '').trim().toUpperCase()
    });
  }
  return list;
}

/**
 * Baca sheet generik sebagai array of object (header = key)
 */
function readSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) obj[headers[j]] = data[i][j];
    rows.push(obj);
  }
  return rows;
}

/**
 * Ambil nama sheet Muadalah berdasarkan kelas + semester.
 * kelas: 'X' | 'XI' | 'XII' (boleh 'X A', akan dinormalisasi)
 * semester: '1'..'6'
 *   1,3,5 atau 'I' / 'ganjil'  → ganjil
 *   2,4,6 atau 'II' / 'genap'  → genap
 */
function getSheetNamaMuadalah_(kelas, semester) {
  var k = String(kelas || '').trim().toUpperCase();
  var kelasBesar = '';
  if (k.indexOf('XII') === 0) kelasBesar = 'XII';
  else if (k.indexOf('XI') === 0) kelasBesar = 'XI';
  else if (k.indexOf('X') === 0) kelasBesar = 'X';
  if (!kelasBesar) return null;

  var smt = String(semester || '').trim();
  var suffix = '';
  if (smt === '1' || smt === '3' || smt === '5' || smt === 'I' || smt.toLowerCase() === 'ganjil') suffix = 'I';
  else if (smt === '2' || smt === '4' || smt === '6' || smt === 'II' || smt.toLowerCase() === 'genap') suffix = 'II';
  else return null;

  if (kelasBesar === 'X')   return suffix === 'I' ? SHEET_NAMES.muadalahX_I   : SHEET_NAMES.muadalahX_II;
  if (kelasBesar === 'XI')  return suffix === 'I' ? SHEET_NAMES.muadalahXI_I  : SHEET_NAMES.muadalahXI_II;
  if (kelasBesar === 'XII') return suffix === 'I' ? SHEET_NAMES.muadalahXII_I : SHEET_NAMES.muadalahXII_II;
  return null;
}

/**
 * Baca semua konfigurasi dari sheet KonfigMuadalah
 */
function readKonfigMuadalah_(ss) {
  var sheet = ss.getSheetByName(SHEET_NAMES.muadalahKonfig);
  if (!sheet) return [];

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

  var lastCol = Math.max(MUADALAH_KONFIG_COL.MAPEL_AKHIR + 1, sheet.getLastColumn());
  var data = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

  var list = [];
  for (var i = 0; i < data.length; i++) {
    var row = data[i];
    var tahun = String(row[MUADALAH_KONFIG_COL.TAHUN_PELAJARAN] || '').trim();
    var kelas = String(row[MUADALAH_KONFIG_COL.KELAS] || '').trim().toUpperCase();

    if (!tahun || !kelas) continue;

    // Kumpulkan mapel dari MAPEL 1 sampai ketemu kosong
    var mapel = [];
    for (var m = MUADALAH_KONFIG_COL.MAPEL_AWAL; m <= MUADALAH_KONFIG_COL.MAPEL_AKHIR; m++) {
      var namaMapel = String(row[m] || '').trim();
      if (!namaMapel) break;
      mapel.push(namaMapel);
    }

    if (mapel.length === 0) continue;

    list.push({
      no: row[MUADALAH_KONFIG_COL.NO] || (i + 1),
      tahunPelajaran: tahun,
      kelas: kelas,
      jumlahMapel: mapel.length,
      mapel: mapel
    });
  }
  return list;
}

/**
 * Cari konfigurasi yang cocok untuk (kelas, tahun, semester)
 */
function getKonfigForKelasTahun_(konfigList, kelas, tahun) {
  var k = String(kelas || '').trim().toUpperCase();
  var kelasBesar = '';
  if (k.indexOf('XII') === 0) kelasBesar = 'XII';
  else if (k.indexOf('XI') === 0) kelasBesar = 'XI';
  else if (k.indexOf('X') === 0) kelasBesar = 'X';

  var t = String(tahun || '').trim();

  for (var i = 0; i < konfigList.length; i++) {
    var c = konfigList[i];
    if (c.kelas === kelasBesar && c.tahunPelajaran === t) {
      return c;
    }
  }
  return null;
}

  }

function getDaftarKelasMuadalah() {
  try {
    var ss = getSpreadsheet();

    var set = {};

    konfigList.forEach(function(c) {
      if (c.kelas) set[c.kelas] = true;

    arr.sort(function(a, b) {

    return { status: 'error', message: e.toString() };
  }

/**
 * Simpan konfigurasi (tambah baru ATAU update).
 * payload = {
 *   tahunPelajaran, kelas, mapel: [],
 *   // OPSIONAL — diisi saat EDIT:
 *   originalTahun, originalKelas
 * }
 */
function saveKonfigMuadalah(payload) {
  try {
    if (!payload || !payload.tahunPelajaran || !payload.kelas) {
      return { status: 'error', message: 'Data tidak lengkap!' };
    }

    var tahun = String(payload.tahunPelajaran).trim();
    var kelas = String(payload.kelas).trim().toUpperCase();
    var mapel = payload.mapel || [];

    if (!Array.isArray(mapel) || mapel.length === 0) {
      return { status: 'error', message: 'Minimal 1 mapel!' };
    }
    if (mapel.length > MUADALAH_MAX_MAPEL) {
      return { status: 'error', message: 'Maksimal ' + MUADALAH_MAX_MAPEL + ' mapel!' };
    }

    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.muadalahKonfig);
    if (!sheet) {
      return { status: 'error', message: 'Sheet KonfigMuadalah tidak ditemukan!' };
    }

    var data = sheet.getDataRange().getValues();

    var isEdit = (payload.originalTahun !== undefined && payload.originalTahun !== null && payload.originalTahun !== '') ||
                 (payload.originalKelas !== undefined && payload.originalKelas !== null && payload.originalKelas !== '');

    var cariTahun, cariKelas;
    if (isEdit) {
      cariTahun = String(payload.originalTahun || '').trim();
      cariKelas = String(payload.originalKelas || '').trim().toUpperCase();
    } else {
      cariTahun = tahun;
      cariKelas = kelas;
    }

    // Cari baris lama
    var baris = -1;
    for (var i = 1; i < data.length; i++) {
      var t = String(data[i][MUADALAH_KONFIG_COL.TAHUN_PELAJARAN] || '').trim();
      var k = String(data[i][MUADALAH_KONFIG_COL.KELAS] || '').trim().toUpperCase();
      if (t === cariTahun && k === cariKelas) {
        baris = i + 1;
        break;
      }
    }

    // Cek duplikat saat TAMBAH BARU
    if (!isEdit) {
      for (var j = 1; j < data.length; j++) {
        var t2 = String(data[j][MUADALAH_KONFIG_COL.TAHUN_PELAJARAN] || '').trim();
        var k2 = String(data[j][MUADALAH_KONFIG_COL.KELAS] || '').trim().toUpperCase();
        if (t2 === tahun && k2 === kelas) {
          return {
            status: 'error',
            message: 'Konfigurasi ' + kelas + ' • ' + tahun + ' sudah ada. Gunakan tombol Edit untuk mengubahnya.'
          };
        }
      }
    }

    // Cek konflik saat EDIT (kalau kunci baru bentrok dengan baris lain)
    if (isEdit) {
      var sudahAda = false;
      for (var m = 1; m < data.length; m++) {
        if (m + 1 === baris) continue;
        var t3 = String(data[m][MUADALAH_KONFIG_COL.TAHUN_PELAJARAN] || '').trim();
        var k3 = String(data[m][MUADALAH_KONFIG_COL.KELAS] || '').trim().toUpperCase();
        if (t3 === tahun && k3 === kelas) {
          sudahAda = true;
          break;
        }
      }
      if (sudahAda) {
        return {
          status: 'error',
          message: 'Sudah ada konfigurasi lain dengan ' + kelas + ' • ' + tahun + '.'
        };
      }
    }

    // Bangun baris
    var row = [];
    row[MUADALAH_KONFIG_COL.TAHUN_PELAJARAN] = tahun;
    row[MUADALAH_KONFIG_COL.KELAS] = kelas;

    for (var x = 0; x < MUADALAH_MAX_MAPEL; x++) {
      row[MUADALAH_KONFIG_COL.MAPEL_AWAL + x] = mapel[x] || '';
    }

    if (baris === -1) {
      row[MUADALAH_KONFIG_COL.NO] = sheet.getLastRow();
      sheet.appendRow(row);
      return { status: 'success', message: 'Konfigurasi baru berhasil disimpan.' };
    } else {
      row[MUADALAH_KONFIG_COL.NO] = data[baris - 1][MUADALAH_KONFIG_COL.NO];
      sheet.getRange(baris, 1, 1, row.length).setValues([row]);
      return { status: 'success', message: 'Konfigurasi berhasil diperbarui.' };
    }
  } catch (e) {
    console.error('Error saveKonfigMuadalah:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Hapus 1 konfigurasi berdasarkan (tahun, kelas).
 */
function hapusKonfigMuadalah(tahun, kelas) {
  try {
    if (!tahun || !kelas) {
      return { status: 'error', message: 'Parameter tidak lengkap!' };
    }

    var targetTahun = String(tahun).trim();
    var targetKelas = String(kelas).trim().toUpperCase();

    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.muadalahKonfig);
    if (!sheet) {
      return { status: 'error', message: 'Sheet KonfigMuadalah tidak ditemukan!' };
    }

    var data = sheet.getDataRange().getValues();
    var baris = -1;
    for (var i = 1; i < data.length; i++) {
      var t = String(data[i][MUADALAH_KONFIG_COL.TAHUN_PELAJARAN] || '').trim();
      var k = String(data[i][MUADALAH_KONFIG_COL.KELAS] || '').trim().toUpperCase();
      if (t === targetTahun && k === targetKelas) {
        baris = i + 1;
        break;
      }
    }

    if (baris === -1) {
      return { status: 'error', message: 'Konfigurasi tidak ditemukan.' };
    }

    sheet.deleteRow(baris);

    // Renumber kolom NO
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var noCol = sheet.getRange(2, MUADALAH_KONFIG_COL.NO + 1, lastRow - 1, 1);
      var noVals = noCol.getValues();
      for (var j = 0; j < noVals.length; j++) noVals[j][0] = j + 1;
      noCol.setValues(noVals);
    }

    return { status: 'success', message: 'Konfigurasi berhasil dihapus.' };
  } catch (e) {
    console.error('Error hapusKonfigMuadalah:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Baca semua nilai dari sheet Muadalah (per kelas + semester).
 */
function readNilaiMuadalah_(ss, kelas, semester) {
  var namaSheet = getSheetNamaMuadalah_(kelas, semester);
  if (!namaSheet) return [];

  var sheet = ss.getSheetByName(namaSheet);
  if (!sheet) return [];

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

  var lastCol = Math.max(MUADALAH_NILAI_COL.RANKING + 1, sheet.getLastColumn());
  var data = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

  var list = [];
  for (var i = 0; i < data.length; i++) {
    var row = data[i];
    var noInduk = String(row[MUADALAH_NILAI_COL.NO_INDUK] || '').trim();
    var nama    = String(row[MUADALAH_NILAI_COL.NAMA] || '').trim();
    if (!noInduk && !nama) continue;

    var nilaiArr = [];
    for (var n = MUADALAH_NILAI_COL.N_AWAL; n <= MUADALAH_NILAI_COL.N_AKHIR; n++) {
      var v = row[n];
      if (v === '' || v === null || v === undefined) nilaiArr.push('');
      else nilaiArr.push(v);
    }

    list.push({
      baris: i + 2,
      noInduk: noInduk,
      nama: nama,
      kelas: String(row[MUADALAH_NILAI_COL.KELAS] || '').trim(),
      tahunPelajaran: String(row[MUADALAH_NILAI_COL.TAHUN_PELAJARAN] || '').trim(),
      nilai: nilaiArr,
      jumlah: row[MUADALAH_NILAI_COL.JUMLAH] || 0,
      rataRata: row[MUADALAH_NILAI_COL.RATA_RATA] || 0,
      ranking: row[MUADALAH_NILAI_COL.RANKING] || 0
    });
  }
  return list;
}

/**
 * Simpan nilai Muadalah (batch).
 * payload = {
 *   kelas, tahunPelajaran, semester,
 *   siswa: [{ noInduk, nama, kelas, nilai: [] }]
 * }
 */
function saveNilaiMuadalah(payload) {
  try {
    if (!payload || !payload.kelas || !payload.tahunPelajaran || !payload.semester) {
      return { status: 'error', message: 'Data tidak lengkap!' };
    }
    if (!Array.isArray(payload.siswa) || payload.siswa.length === 0) {
      return { status: 'error', message: 'Tidak ada siswa yang disimpan!' };
    }

    var ss = getSpreadsheet();
    var kelas = String(payload.kelas).trim();
    var tahun = String(payload.tahunPelajaran).trim();
    var smt = String(payload.semester).trim();

    // Cari nama sheet dari kelas + semester
    var namaSheet = getSheetNamaMuadalah_(kelas, smt);
    if (!namaSheet) {
      return { status: 'error', message: 'Kelas/semester tidak valid: ' + kelas + ' / ' + smt };
    }

    var sheet = ss.getSheetByName(namaSheet);
    if (!sheet) {
      return { status: 'error', message: 'Sheet ' + namaSheet + ' tidak ditemukan!' };
    }

    // Konfigurasi mapel (tanpa semester)
    var konfigList = readKonfigMuadalah_(ss);
    var konfig = getKonfigForKelasTahun_(konfigList, kelas, tahun);
    if (!konfig) {
      return { status: 'error', message: 'Konfigurasi tidak ditemukan untuk kelas/tahun ini.' };
    }

    var jumlahMapel = konfig.jumlahMapel;

    // Baca data existing
    var lastRow = sheet.getLastRow();
    var data = lastRow > 1
      ? sheet.getRange(2, 1, lastRow - 1, MUADALAH_NILAI_COL.RANKING + 1).getValues()
      : [];

    // Index per No Induk
    var indexMap = {};
    for (var i = 0; i < data.length; i++) {
      var ni = String(data[i][MUADALAH_NILAI_COL.NO_INDUK] || '').trim();
      if (ni) indexMap[ni] = i;
    }

    var ditambah = 0;
    var diupdate = 0;

    payload.siswa.forEach(function(item) {
      var ni = String(item.noInduk || '').trim();
      var namaItem = String(item.nama || '').trim();
      var kelasItem = String(item.kelas || kelas).trim();
      if (!ni) return;

      var idx = indexMap[ni];

      var nilaiArr = new Array(MUADALAH_MAX_MAPEL);
      for (var m = 0; m < MUADALAH_MAX_MAPEL; m++) {
        if (m < jumlahMapel) {
          var val = item.nilai && item.nilai[m] !== undefined ? item.nilai[m] : '';
          nilaiArr[m] = val;
        } else {
          nilaiArr[m] = '';
        }
      }

      if (idx === undefined) {
        // Baris baru
        var newRow = new Array(MUADALAH_NILAI_COL.RANKING + 1).fill('');
        newRow[MUADALAH_NILAI_COL.NO_INDUK] = ni;
        newRow[MUADALAH_NILAI_COL.NAMA] = namaItem;
        newRow[MUADALAH_NILAI_COL.KELAS] = kelasItem;
        newRow[MUADALAH_NILAI_COL.TAHUN_PELAJARAN] = tahun;
        for (var m2 = 0; m2 < MUADALAH_MAX_MAPEL; m2++) {
          newRow[MUADALAH_NILAI_COL.N_AWAL + m2] = nilaiArr[m2];
        }
        data.push(newRow);
        indexMap[ni] = data.length - 1;
        ditambah++;
      } else {
        // Update
        data[idx][MUADALAH_NILAI_COL.NO_INDUK] = ni;
        data[idx][MUADALAH_NILAI_COL.NAMA] = namaItem;
        data[idx][MUADALAH_NILAI_COL.KELAS] = kelasItem;
        data[idx][MUADALAH_NILAI_COL.TAHUN_PELAJARAN] = tahun;
        for (var m3 = 0; m3 < MUADALAH_MAX_MAPEL; m3++) {
          data[idx][MUADALAH_NILAI_COL.N_AWAL + m3] = nilaiArr[m3];
        }
        diupdate++;
      }
    });

    if (data.length > 0) {
      var maxCol = MUADALAH_NILAI_COL.RANKING + 1;
      for (var j = 0; j < data.length; j++) {
        while (data[j].length < maxCol) data[j].push('');
      }
      sheet.getRange(2, 1, data.length, maxCol).setValues(data);
    }

    SpreadsheetApp.flush();

    var hasilHitung = hitungUlangJumlahRataRanking_(ss, kelas, smt, tahun);

    return {
      status: 'success',
      message: 'Berhasil disimpan. Ditambah: ' + ditambah + ', diperbarui: ' + diupdate + '.',
      ditambah: ditambah,
      diupdate: diupdate,
      hitung: hasilHitung
    };

  } catch (e) {
    console.error('Error saveNilaiMuadalah:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Hitung ulang JUMLAH, RATA-RATA, RANKING untuk (kelas, semester, tahun).
 */
function hitungUlangJumlahRataRanking_(ss, kelas, smt, tahun) {
  var namaSheet = getSheetNamaMuadalah_(kelas, smt);
  if (!namaSheet) return { status: 'error', message: 'Kelas/semester tidak valid' };

  var sheet = ss.getSheetByName(namaSheet);
  if (!sheet) return { status: 'error', message: 'Sheet tidak ditemukan' };

  var konfigList = readKonfigMuadalah_(ss);
  var konfig = getKonfigForKelasTahun_(konfigList, kelas, tahun);
  if (!konfig) return { status: 'error', message: 'Konfigurasi tidak ditemukan' };

  var jumlahMapel = konfig.jumlahMapel;

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return { status: 'success', total: 0 };

  var lastCol = MUADALAH_NILAI_COL.RANKING + 1;
  var data = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

  var targetTahun = String(tahun).trim();
  var siswaList = [];

  for (var i = 0; i < data.length; i++) {
    var tp = String(data[i][MUADALAH_NILAI_COL.TAHUN_PELAJARAN] || '').trim();
    if (tp !== targetTahun) continue;

    siswaList.push({
      idx: i,
      noInduk: String(data[i][MUADALAH_NILAI_COL.NO_INDUK] || '').trim(),
      nama: String(data[i][MUADALAH_NILAI_COL.NAMA] || '').trim(),
      kelas: String(data[i][MUADALAH_NILAI_COL.KELAS] || '').trim()
    });
  }

  if (siswaList.length === 0) return { status: 'success', total: 0 };

  // Hitung JUMLAH & RATA-RATA
  siswaList.forEach(function(s) {
    var row = data[s.idx];
    var total = 0;
    var count = 0;

    for (var m = 0; m < jumlahMapel; m++) {
      var v = row[MUADALAH_NILAI_COL.N_AWAL + m];
      if (v === '' || v === null || v === undefined) continue;

      var num = parseFloat(v);
      if (isNaN(num)) continue;

      total += num;
      count++;
    }

    var rata = count > 0 ? (total / count) : 0;

    s.total = total;
    s.count = count;
    s.rata = rata;

    data[s.idx][MUADALAH_NILAI_COL.JUMLAH] = count > 0 ? Math.round(total * 100) / 100 : '';
    data[s.idx][MUADALAH_NILAI_COL.RATA_RATA] = count > 0 ? Math.round(rata * 100) / 100 : '';
  });

  // Ranking per kelas paralel
  var grupKelas = {};
  siswaList.forEach(function(s) {
    var k = s.kelas || '(tanpa kelas)';
    if (!grupKelas[k]) grupKelas[k] = [];
    grupKelas[k].push(s);
  });

  Object.keys(grupKelas).forEach(function(kelasParalel) {
    var grup = grupKelas[kelasParalel];
    grup.sort(function(a, b) { return b.rata - a.rata; });

    var currentRank = 1;
    for (var i = 0; i < grup.length; i++) {
      if (i > 0 && grup[i].rata === grup[i - 1].rata) {
        data[grup[i].idx][MUADALAH_NILAI_COL.RANKING] = data[grup[i - 1].idx][MUADALAH_NILAI_COL.RANKING];
      } else {
        data[grup[i].idx][MUADALAH_NILAI_COL.RANKING] = currentRank;
      }
      currentRank = i + 2;
    }
  });

  sheet.getRange(2, 1, data.length, lastCol).setValues(data);

  return {
    status: 'success',
    total: siswaList.length,
    message: 'JUMLAH/RATA-RATA/RANKING diperbarui untuk ' + siswaList.length + ' siswa.'
  };
}

/**
 * Ambil konfigurasi surat (jabatan, bagian, jenis, lembaga, pesantren).
 */
function getKonfigSurat() {
  try {
    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.konfigSurat);
    if (!sheet) return { status: 'error', message: 'Sheet KonfigSurat tidak ditemukan!' };

    var data = sheet.getDataRange().getValues();
    if (data.length < 2) return { status: 'error', message: 'KonfigSurat kosong!' };

    var jabatanList = [];
    var bagianList = [];
    var jenisList = [];
    var kodeLembaga = '';
    var kodePesantren = '';
    var seenJabatan = {}, seenBagian = {}, seenJenis = {};

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var kJab = String(row[KONFIG_SURAT_COL.KODE_JABATAN] || '').trim();
      var jab  = String(row[KONFIG_SURAT_COL.JABATAN] || '').trim();
      var kBag = String(row[KONFIG_SURAT_COL.KODE_BAGIAN] || '').trim();
      var bag  = String(row[KONFIG_SURAT_COL.BAGIAN_SURAT] || '').trim();
      var kJen = String(row[KONFIG_SURAT_COL.KODE_JENIS] || '').trim();
      var jen  = String(row[KONFIG_SURAT_COL.JENIS_SURAT] || '').trim();
      var kLem = String(row[KONFIG_SURAT_COL.KODE_LEMBAGA] || '').trim();
      var kPes = String(row[KONFIG_SURAT_COL.KODE_PESANTREN] || '').trim();

      if (kJab && jab && !seenJabatan[kJab]) {
        seenJabatan[kJab] = true;
        jabatanList.push({ kode: kJab, nama: jab });
      }
      if (kBag && bag && !seenBagian[kBag]) {
        seenBagian[kBag] = true;
        bagianList.push({ kode: kBag, nama: bag });
      }
      if (kJen && jen && !seenJenis[kJen]) {
        seenJenis[kJen] = true;
        jenisList.push({ kode: kJen, nama: jen });
      }
      if (kLem && !kodeLembaga) kodeLembaga = kLem;
      if (kPes && !kodePesantren) kodePesantren = kPes;
    }

    return {
      status: 'success',
      jabatan: jabatanList,
      bagian: bagianList,
      jenis: jenisList,
      kodeLembaga: kodeLembaga || 'SMAI',
      kodePesantren: kodePesantren || 'DUBA'
    };
  } catch (e) {
    console.error('Error getKonfigSurat:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Cari jabatan berdasarkan ROLE user login.
 * Cocokkan longgar (contains, case-insensitive).
 * Contoh: role "kurikulum" → jabatan "Wakil Kurikulum" / "Kurikulum"
 */
function getJabatanByRole(role) {
  try {
    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.konfigSurat);
    if (!sheet) return null;

    var data = sheet.getDataRange().getValues();
    var r = String(role || '').trim().toLowerCase();
    if (!r) return null;

    for (var i = 1; i < data.length; i++) {
      var kJab = String(data[i][KONFIG_SURAT_COL.KODE_JABATAN] || '').trim();
      var jab  = String(data[i][KONFIG_SURAT_COL.JABATAN] || '').trim();
      var jabLower = jab.toLowerCase();

      if (jabLower === r || jabLower.indexOf(r) !== -1 || r.indexOf(jabLower) !== -1) {
        return { kode: kJab, nama: jab };
      }
    }
    return null;
  } catch (e) {
    console.error('Error getJabatanByRole:', e);
    return null;
  }
}

/**
 * Ambil nomor urut berikutnya untuk bulan & tahun tertentu.
 * Lihat di RiwayatSurat, cari nomor tertinggi dengan bulan+tahun yang sama.
 */
function getNomorUrutBerikutnya_(sheet, bulan, tahun) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return 1;

  var data = sheet.getRange(2, 1, lastRow - 1, RIWAYAT_SURAT_COL.TAHUN + 1).getValues();
  var maxNo = 0;

  for (var i = 0; i < data.length; i++) {
    var b = String(data[i][RIWAYAT_SURAT_COL.BULAN] || '').trim();
    var t = String(data[i][RIWAYAT_SURAT_COL.TAHUN] || '').trim();
    var nomor = String(data[i][RIWAYAT_SURAT_COL.NOMOR] || '').trim();

    if (b === String(bulan) && t === String(tahun)) {
      // Ambil angka pertama dari nomor surat (misal "11/K.01/..." → 11)
      var m = nomor.match(/^(\d+)/);
      if (m) {
        var n = parseInt(m[1], 10);
        if (!isNaN(n) && n > maxNo) maxNo = n;
      }
    }
  }
  return maxNo + 1;
}

/**
 * Terbitkan nomor surat baru.
 * payload = { username, role, kodeBagian, kodeJenis, perihal }
 */
function terbitkanNomorSurat(payload) {
  try {
    if (!payload || !payload.kodeBagian || !payload.kodeJenis) {
      return { status: 'error', message: 'Bagian dan Jenis Surat harus dipilih!' };
    }

    var ss = getSpreadsheet();

    // === Konfigurasi ===
    var konfig = getKonfigSurat();
    if (konfig.status !== 'success') return konfig;

    // === Jabatan dari ROLE user login ===
    var jabatan = getJabatanByRole(payload.role);
    if (!jabatan) {
      return { status: 'error', message: 'Jabatan untuk role "' + payload.role + '" tidak ditemukan di KonfigSurat.' };
    }

    // === Bagian & Jenis ===
    var bagian = konfig.bagian.find(function(b) { return b.kode === payload.kodeBagian; });
    var jenis  = konfig.jenis.find(function(j)  { return j.kode === payload.kodeJenis; });
    if (!bagian) return { status: 'error', message: 'Bagian tidak valid.' };
    if (!jenis)  return { status: 'error', message: 'Jenis Surat tidak valid.' };

    // === Bulan Romawi & Tahun ===
    var bulanIndo = ['Januari','Februari','Maret','April','Mei','Juni','Juli','Agustus','September','Oktober','November','Desember'];
    var bulanRomawi = ['I','II','III','IV','V','VI','VII','VIII','IX','X','XI','XII'];
    var now = new Date();
    var bulan = bulanIndo[now.getMonth()];
    var bulanRom = bulanRomawi[now.getMonth()];
    var tahun = String(now.getFullYear());

    // === Sheet RiwayatSurat ===
    var sheetR = ss.getSheetByName(SHEET_NAMES.riwayatSurat);
    if (!sheetR) return { status: 'error', message: 'Sheet RiwayatSurat tidak ditemukan!' };

        // === Nomor urut ===
    var nomorUrut = getNomorUrutBerikutnya_(sheetR, bulan, tahun);

    // === Pad nomor jadi 4 digit (0001, 0002, ...) ===
    var nomorPad = ('0000' + nomorUrut).slice(-4);

    // === Susun nomor surat ===
    // Format: nomor/jabatan/bagian.jenis/lembaga/pesantren/bulan-romawi/tahun
    var nomorSurat = nomorPad + '/' +
      jabatan.kode + '/' +
      bagian.kode + '.' + jenis.kode + '/' +
      konfig.kodeLembaga + '/' +
      konfig.kodePesantren + '/' +
      bulanRom + '/' +
      tahun;

    // === Simpan ke RiwayatSurat ===
    var tanggalStr = now.getDate() + ' ' + bulan + ' ' + tahun;
    sheetR.appendRow([
      tanggalStr,
      nomorSurat,
      jabatan.nama,
      bagian.nama,
      jenis.nama,
      konfig.kodeLembaga,
      konfig.kodePesantren,
      bulan,
      tahun,
      payload.perihal || jenis.nama
    ]);

    return {
      status: 'success',
      nomorSurat: nomorSurat,
      nomorUrut: nomorUrut,
      tanggal: tanggalStr,
      jabatan: jabatan,
      bagian: bagian,
      jenis: jenis,
      bulan: bulan,
      tahun: tahun
    };
  } catch (e) {
    console.error('Error terbitkanNomorSurat:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Simpan konfigurasi surat.
 * payload = {
 *   kodeLembaga, kodePesantren,
 *   jabatanList: [{kode, nama}],
 *   bagianList:  [{kode, nama}],
 *   jenisList:   [{kode, nama}]
 * }
 * Setiap kombinasi unik akan di-flatten jadi baris di sheet.
 */
function saveKonfigSurat(payload) {
  try {
    if (!payload) return { status: 'error', message: 'Data tidak valid!' };

    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.konfigSurat);
    if (!sheet) return { status: 'error', message: 'Sheet KonfigSurat tidak ditemukan!' };

    var kodeLembaga = String(payload.kodeLembaga || '').trim();
    var kodePesantren = String(payload.kodePesantren || '').trim();
    if (!kodeLembaga || !kodePesantren) {
      return { status: 'error', message: 'Kode Lembaga dan Pesantren wajib diisi!' };
    }

    var jabatanList = payload.jabatanList || [];
    var bagianList = payload.bagianList || [];
    var jenisList = payload.jenisList || [];

    if (jabatanList.length === 0 && bagianList.length === 0 && jenisList.length === 0) {
      return { status: 'error', message: 'Minimal isi 1 data (Jabatan / Bagian / Jenis)!' };
    }

    // Header
    var header = ['KODE JABATAN', 'JABATAN', 'KODE BAGIAN', 'BAGIAN SURAT', 'KODE JENIS SURAT', 'JENIS SURAT', 'KODE LEMBAGA', 'KODE PESANTREN'];
    var data = [header];

    // Flatten: setiap kombinasi unik jadi 1 baris
    var maxLen = Math.max(jabatanList.length, bagianList.length, jenisList.length, 1);
    for (var i = 0; i < maxLen; i++) {
      var jab = jabatanList[i] || { kode: '', nama: '' };
      var bag = bagianList[i]  || { kode: '', nama: '' };
      var jen = jenisList[i]   || { kode: '', nama: '' };

      data.push([
        String(jab.kode || '').trim(),
        String(jab.nama || '').trim(),
        String(bag.kode || '').trim(),
        String(bag.nama || '').trim(),
        String(jen.kode || '').trim(),
        String(jen.nama || '').trim(),
        kodeLembaga,
        kodePesantren
      ]);
    }

    sheet.clearContents();
    sheet.getRange(1, 1, data.length, header.length).setValues(data);

    return {
      status: 'success',
      message: 'Konfigurasi tersimpan. Jabatan: ' + jabatanList.length +
               ', Bagian: ' + bagianList.length +
               ', Jenis: ' + jenisList.length + '.'
    };
  } catch (e) {
    console.error('Error saveKonfigSurat:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Ambil data mentah KonfigSurat untuk diedit di UI.
 * Return: { kodeLembaga, kodePesantren, jabatanList, bagianList, jenisList }
 */
function getKonfigSuratRaw() {
  try {
    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.konfigSurat);
    if (!sheet) return { status: 'error', message: 'Sheet KonfigSurat tidak ditemukan!' };

    var data = sheet.getDataRange().getValues();
    var jabatanSet = {}, bagianSet = {}, jenisSet = {};
    var kodeLembaga = '', kodePesantren = '';

    for (var i = 1; i < data.length; i++) {
      var row = data[i];

      var kJab = String(row[KONFIG_SURAT_COL.KODE_JABATAN] || '').trim();
      var jab  = String(row[KONFIG_SURAT_COL.JABATAN] || '').trim();
      var kBag = String(row[KONFIG_SURAT_COL.KODE_BAGIAN] || '').trim();
      var bag  = String(row[KONFIG_SURAT_COL.BAGIAN_SURAT] || '').trim();
      var kJen = String(row[KONFIG_SURAT_COL.KODE_JENIS] || '').trim();
      var jen  = String(row[KONFIG_SURAT_COL.JENIS_SURAT] || '').trim();

      if (kJab && jab && !jabatanSet[kJab]) jabatanSet[kJab] = { kode: kJab, nama: jab };
      if (kBag && bag && !bagianSet[kBag]) bagianSet[kBag] = { kode: kBag, nama: bag };
      if (kJen && jen && !jenisSet[kJen]) jenisSet[kJen] = { kode: kJen, nama: jen };

      if (!kodeLembaga) kodeLembaga = String(row[KONFIG_SURAT_COL.KODE_LEMBAGA] || '').trim();
      if (!kodePesantren) kodePesantren = String(row[KONFIG_SURAT_COL.KODE_PESANTREN] || '').trim();
    }

    function toArr(set) {
      return Object.keys(set).map(function(k) { return set[k]; });
    }

    return {
      status: 'success',
      kodeLembaga: kodeLembaga || 'SMAI',
      kodePesantren: kodePesantren || 'DUBA',
      jabatanList: toArr(jabatanSet),
      bagianList: toArr(bagianSet),
      jenisList: toArr(jenisSet)
    };
  } catch (e) {
    console.error('Error getKonfigSuratRaw:', e);
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Ambil riwayat surat (untuk ditampilkan di UI).
 */
function getRiwayatSurat() {
  try {
    var ss = getSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAMES.riwayatSurat);
    if (!sheet) return { status: 'success', data: [] };

    var data = sheet.getDataRange().getValues();
    if (data.length < 2) return { status: 'success', data: [] };

    var list = [];
    for (var i = data.length - 1; i >= 1; i--) {
      var row = data[i];
      if (!row[RIWAYAT_SURAT_COL.NOMOR]) continue;
      list.push({
        tanggal: String(row[RIWAYAT_SURAT_COL.TANGGAL] || ''),
        nomor: String(row[RIWAYAT_SURAT_COL.NOMOR] || ''),
        diterbitkanOleh: String(row[RIWAYAT_SURAT_COL.DITERBITKAN_OLEH] || ''),
        bagian: String(row[RIWAYAT_SURAT_COL.BAGIAN] || ''),
        jenisSurat: String(row[RIWAYAT_SURAT_COL.JENIS_SURAT] || ''),
        lembaga: String(row[RIWAYAT_SURAT_COL.LEMBAGA] || ''),
        pondok: String(row[RIWAYAT_SURAT_COL.PONDOK] || ''),
        bulan: String(row[RIWAYAT_SURAT_COL.BULAN] || ''),
        tahun: String(row[RIWAYAT_SURAT_COL.TAHUN] || ''),
        perihal: String(row[RIWAYAT_SURAT_COL.PERIHAL] || '')
      });
    }

    return { status: 'success', data: list };
  } catch (e) {
    console.error('Error getRiwayatSurat:', e);
    return { status: 'error', message: e.toString() };
  }
}
